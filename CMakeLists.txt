cmake_minimum_required(VERSION 3.25)
project(retinex_gpu VERSION 2.1 LANGUAGES CXX)
option(RETINEX_CUDA "Include both NVIDIA CUDA engines" ON)
option(RETINEX_TESTING "Enable failure injection for tests only" OFF)
if(RETINEX_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75-real 86-real 89-real 120-real 75-virtual)
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit 12.8 REQUIRED)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY Static)
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_FOUND)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c
    "import importlib.util,pathlib; s=importlib.util.find_spec('vapoursynth'); print(pathlib.Path(s.origin).parent/'include' if s else '')"
    OUTPUT_VARIABLE VS_PY_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
find_path(VAPOURSYNTH_INCLUDE_DIR VapourSynth4.h HINTS
  "${VS_PY_INCLUDE}" "$ENV{VAPOURSYNTH_SDK}/include" "$ENV{VAPOURSYNTH_ROOT}/include"
  "$ENV{ProgramFiles}/VapourSynth/sdk/include" "$ENV{ProgramFiles}/VapourSynth/include")
if(NOT VAPOURSYNTH_INCLUDE_DIR)
  message(FATAL_ERROR "VapourSynth4.h not found. Set -DVAPOURSYNTH_INCLUDE_DIR=path/to/include")
endif()
add_library(retinex_gpu SHARED src/plugin.cpp src/exact/cpu.cpp src/fast/cpu.cpp)
target_include_directories(retinex_gpu PRIVATE "${VAPOURSYNTH_INCLUDE_DIR}")
if(RETINEX_CUDA)
  target_sources(retinex_gpu PRIVATE src/exact/gpu.cu src/fast/gpu.cu)
  target_compile_definitions(retinex_gpu PRIVATE RETINEX_FAST_CUDA)
  target_link_libraries(retinex_gpu PRIVATE CUDA::cudart_static)
endif()
if(RETINEX_TESTING)
  target_compile_definitions(retinex_gpu PRIVATE RETINEX_FAST_TESTING)
endif()
target_compile_definitions(retinex_gpu PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
# Reference double recurrences are sensitive to reassociation and fused multiply-add.
target_compile_options(retinex_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false>)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  target_compile_options(retinex_gpu PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/fp:precise>)
  set_property(TARGET retinex_gpu PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(retinex_gpu PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-ffp-contract=off>)
endif()
set_target_properties(retinex_gpu PROPERTIES PREFIX "")
install(TARGETS retinex_gpu RUNTIME DESTINATION . LIBRARY DESTINATION .)
option(BUILD_CPU_REFERENCES "Build downloaded upstream CPU plugins for verification" OFF)
if(BUILD_CPU_REFERENCES)
  add_subdirectory(tests/reference)
endif()
